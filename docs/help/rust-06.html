6. จัดการข้อผิดพลาด

panic, Result และเครื่องหมาย ?

สองแบบของข้อผิดพลาด

  • ข้อผิดพลาดที่คาดได้ เช่น ไฟล์ไม่มี ผู้ใช้พิมพ์ตัวเลขผิด เน็ตหลุด ควรจัดการและบอกผู้ใช้ ใช้ Result
  • บั๊ก สิ่งที่ไม่ควรเกิดเลยถ้าโค้ดถูก เช่น อ่านตำแหน่งเกินขนาดรายการ Rust จะ panic คือหยุดโปรแกรมพร้อมบอกว่าเกิดที่ไหน

panic

let list = vec![1, 2, 3];
let x = list[10];   // panic: index out of bounds: the len is 3 but the index is 10
panic!("ไม่ควรมาถึงตรงนี้");  // สั่ง panic เอง

ใน Phai Studio ถ้ารันด้วย F5 ดีบักเกอร์จะหยุดตรงจุดที่ panic ให้ดูค่าตัวแปรและ Call Stack ได้ทันที

Result: สำเร็จ หรือ ผิดพลาด

enum Result<T, E> {   // มีอยู่แล้วในภาษา
    Ok(T),             // สำเร็จ ได้ค่า T
    Err(E),            // ผิดพลาด ได้รายละเอียด E
}
fn main() {
    let input = "12a";
    match input.parse::<i32>() {
        Ok(n) => println!("ได้ตัวเลข {n}"),
        Err(e) => println!("ไม่ใช่ตัวเลข: {e}"),
    }
}

เขียนฟังก์ชันที่คืน Result เอง

fn withdraw(balance: f64, amount: f64) -> Result<f64, String> {
    if amount <= 0.0 {
        return Err("จำนวนเงินต้องมากกว่าศูนย์".to_string());
    }
    if amount > balance {
        return Err(format!("เงินไม่พอ มี {balance} แต่จะถอน {amount}"));
    }
    Ok(balance - amount)
}

เครื่องหมาย ? : ส่งต่อข้อผิดพลาด

เมื่อฟังก์ชันของเราเองก็คืน Result ใส่ ? หลังสิ่งที่อาจผิดพลาด ถ้าผิด มันจะคืน Err ออกจากฟังก์ชันทันที ถ้าสำเร็จ ได้ค่าข้างในมาใช้ต่อ

use std::fs;

fn read_total(path: &str) -> Result<i32, Box<dyn std::error::Error>> {
    let text = fs::read_to_string(path)?;     // ไฟล์ไม่มี? คืน Err เลย
    let mut total = 0;
    for line in text.lines() {
        total += line.trim().parse::<i32>()?;  // ไม่ใช่ตัวเลข? คืน Err เลย
    }
    Ok(total)
}

fn main() {
    match read_total("numbers.txt") {
        Ok(t) => println!("รวม {t}"),
        Err(e) => println!("อ่านไม่ได้: {e}"),
    }
}
Box<dyn std::error::Error> แปลว่า "ข้อผิดพลาดชนิดไหนก็ได้" สะดวกเวลาฟังก์ชันอาจผิดได้หลายแบบ

unwrap และ expect

let n: i32 = "42".parse().unwrap();                 // ผิดเมื่อไรก็ panic
let n: i32 = "42".parse().expect("ต้องเป็นตัวเลข");  // panic พร้อมข้อความที่เราเขียน

ใช้ได้ตอนทดลองหรือเมื่อมั่นใจว่าไม่มีทางผิด แต่ในโปรแกรมจริงที่รับข้อมูลจากผู้ใช้ ควรใช้ match, if let หรือ ? แทน

วิธีที่ใช้บ่อย

เขียนได้อะไร
x.unwrap_or(0)ค่าข้างใน หรือ 0 ถ้าผิด/ไม่มี
x.unwrap_or_default()ค่าข้างใน หรือค่าว่างของชนิดนั้น
x.is_ok(), x.is_err()สำเร็จไหม
x.ok()แปลง Result เป็น Option
x.map_err(|e| e.to_string())เปลี่ยนชนิดของข้อผิดพลาด

  2. เขียน fn divide(a: f64, b: f64) -> Result<f64, String> ที่คืน Err เมื่อ b เป็นศูนย์
