8. trait และ generic

เขียนโค้ดครั้งเดียว ใช้ได้กับหลายชนิด

trait: ความสามารถที่หลายชนิดมีร่วมกัน

trait คล้าย interface ในภาษาอื่น บอกว่า "ชนิดใดที่มี trait นี้ ต้องทำสิ่งเหล่านี้ได้"

trait Describe {
    fn describe(&self) -> String;

    // method ที่มีค่าเริ่มต้น ชนิดไหนไม่เขียนเองก็ได้อันนี้
    fn shout(&self) -> String {
        self.describe().to_uppercase()
    }
}

struct Cat { name: String }
struct Car { brand: String, year: u32 }

impl Describe for Cat {
    fn describe(&self) -> String {
        format!("แมวชื่อ {}", self.name)
    }
}

impl Describe for Car {
    fn describe(&self) -> String {
        format!("รถ {} ปี {}", self.brand, self.year)
    }
}

รับอะไรก็ได้ที่มี trait

fn show(item: &impl Describe) {
    println!("{}", item.describe());
}

fn main() {
    show(&Cat { name: "ส้ม".into() });
    show(&Car { brand: "ไผ่มอเตอร์".into(), year: 2024 });

    // รายการที่ปนหลายชนิด: ใช้ Box<dyn Trait>
    let things: Vec<Box<dyn Describe>> = vec![
        Box::new(Cat { name: "ดำ".into() }),
        Box::new(Car { brand: "ทดลอง".into(), year: 2020 }),
    ];
    for t in &things {
        println!("{}", t.describe());
    }
}

trait ที่ Rust มีให้และใช้บ่อย

traitทำให้ทำอะไรได้ได้มาด้วย
Debugพิมพ์ด้วย {:?}#[derive(Debug)]
Clone.clone() สำเนา#[derive(Clone)]
PartialEqเทียบด้วย ==#[derive(PartialEq)]
DefaultDefault::default() ค่าเริ่มต้น#[derive(Default)]
Displayพิมพ์ด้วย {}เขียน impl เอง
use std::fmt;

#[derive(Debug, Clone, PartialEq)]
struct Money(f64);

impl fmt::Display for Money {
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        write!(f, "{:.2} บาท", self.0)
    }
}
// println!("{}", Money(45.5));  →  45.50 บาท

generic: ชนิดเป็นตัวแปร

// T คือชนิดอะไรก็ได้ ที่เทียบมากน้อยได้ (PartialOrd) และคัดลอกได้ (Copy)
fn largest<T: PartialOrd + Copy>(list: &[T]) -> Option<T> {
    let mut best = *list.first()?;
    for &x in list {
        if x > best {
            best = x;
        }
    }
    Some(best)
}

fn main() {
    println!("{:?}", largest(&[3, 9, 2]));        // Some(9)
    println!("{:?}", largest(&[1.5, 0.2]));       // Some(1.5)
    println!("{:?}", largest(&['ก', 'ฮ', 'ม']));  // Some('ฮ')
}

Vec<T>, Option<T>, Result<T, E> ที่ใช้มาตลอดก็คือ generic

Lifetime (รู้จักไว้ก่อน)

บางครั้งจะเห็น &'a str เครื่องหมาย 'a คือ lifetime บอกคอมไพเลอร์ว่าค่าที่ยืมมาอยู่ได้นานแค่ไหน ส่วนใหญ่ Rust เดาให้เอง เมื่อจำเป็นต้องเขียน คอมไพเลอร์จะบอกพร้อมตัวอย่าง มือใหม่ยังไม่ต้องกังวลเรื่องนี้

// คืนข้อความที่ยาวกว่า: ผลลัพธ์อยู่ได้นานเท่ากับ a และ b
fn longer<'a>(a: &'a str, b: &'a str) -> &'a str {
    if a.chars().count() >= b.chars().count() { a } else { b }
}
