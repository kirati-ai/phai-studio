7. Vec, String, HashMap

เก็บข้อมูลหลายชิ้น และเครื่องมือจัดการรายการ

Vec: รายการที่ยืดหดได้

let mut scores: Vec<i32> = Vec::new();
scores.push(80);
scores.push(65);

let names = vec!["ก้อง", "นิด", "ต้น"];   // สร้างพร้อมค่า

println!("มี {} คน", names.len());
println!("คนแรก {}", names[0]);         // เกินขนาด = panic
if let Some(n) = names.get(5) {        // ปลอดภัย ได้ Option
    println!("{n}");
}

for s in &scores {                     // วนแบบยืม รายการยังอยู่
    println!("{s}");
}
scores.sort();
scores.retain(|s| *s >= 70);           // เก็บเฉพาะที่ผ่านเงื่อนไข
let last = scores.pop();               // เอาตัวท้ายออก ได้ Option

String กับ &str

String&str
คืออะไรข้อความที่เป็นเจ้าของเอง แก้ไขได้มุมมองไปยังข้อความที่มีอยู่แล้ว อ่านอย่างเดียว
ได้มาจากString::from(".."), "..".to_string(), format!(..)"ข้อความในเครื่องหมายคำพูด", &my_string
ใช้เป็นพารามิเตอร์เมื่อฟังก์ชันต้องเก็บข้อความไว้เมื่อแค่อ่าน (ส่วนใหญ่)
let mut s = String::from("สวัสดี");
s.push_str(" ครับ");
s.push('!');
let full = format!("{s} วันนี้ {}", 18);

println!("{}", s.contains("ครับ"));     // true
println!("{}", s.replace("ครับ", "ค่ะ"));
for word in "ไผ่ สตูดิโอ ฟรี".split(' ') {
    println!("{word}");
}
let trimmed = "  มีช่องว่าง  ".trim();
ข้อความใน Rust เก็บแบบ UTF-8 อักษรไทยหนึ่งตัวใช้ 3 ไบต์ s.len() จึงนับไบต์ ไม่ใช่ตัวอักษร นับตัวอักษรใช้ s.chars().count() และอย่าตัดข้อความด้วย &s[0..2] กับภาษาไทย เพราะอาจตัดกลางตัวอักษรแล้ว panic ใช้ s.chars().take(2).collect::<String>() แทน

HashMap: เก็บคู่ คีย์ → ค่า

use std::collections::HashMap;

let mut stock: HashMap<String, u32> = HashMap::new();
stock.insert("กาแฟ".to_string(), 10);
stock.insert("ชา".to_string(), 4);

if let Some(n) = stock.get("กาแฟ") {
    println!("กาแฟเหลือ {n}");
}

// นับคำ: ถ้ายังไม่มีให้เริ่มที่ 0 แล้วบวก 1
let mut counts = HashMap::new();
for word in "ชา กาแฟ ชา ชา".split(' ') {
    *counts.entry(word).or_insert(0) += 1;
}

for (name, n) in &stock {
    println!("{name}: {n}");
}

Iterator และ closure

closure คือฟังก์ชันสั้น ๆ ไม่มีชื่อ เขียนแบบ |x| x * 2 ใช้คู่กับ iterator เพื่อแปลงและกรองรายการได้สั้นและอ่านง่าย

let prices = vec![120, 45, 300, 80];

let total: i32 = prices.iter().sum();
let expensive: Vec<i32> = prices.iter().filter(|p| **p > 100).cloned().collect();
let with_vat: Vec<f64> = prices.iter().map(|p| *p as f64 * 1.07).collect();
let max = prices.iter().max();                    // Option
let any_free = prices.iter().any(|p| *p == 0);    // false
let position = prices.iter().position(|p| *p == 300);  // Some(2)
ถ้ายังไม่ถนัด iterator เขียนด้วย for ธรรมดาก็ได้ผลเหมือนกัน ค่อย ๆ เปลี่ยนเมื่อคุ้นแล้ว
  1. รับข้อความยาวหนึ่งย่อหน้า นับว่าแต่ละคำปรากฏกี่ครั้งด้วย HashMap แล้วพิมพ์คำที่ใช้บ่อยที่สุด
  2. จากรายการคะแนน หาค่าเฉลี่ย คะแนนสูงสุด และจำนวนคนที่ได้มากกว่า 50 ด้วย iterator
