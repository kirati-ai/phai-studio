3. เงื่อนไขและการวนซ้ำ

ให้โปรแกรมเลือกทำ และทำซ้ำ

if / else

let temp = 36;
if temp > 37 {
    println!("มีไข้");
} else if temp < 35 {
    println!("ตัวเย็นเกินไป");
} else {
    println!("ปกติ");
}

เงื่อนไขต้องเป็น bool เท่านั้น if count { } แบบภาษาอื่นใช้ไม่ได้ ต้องเขียน if count != 0 { }

ตัวเปรียบเทียบ: == เท่ากับ, != ไม่เท่า, < > <= >= และเชื่อมเงื่อนไขด้วย && (และ), || (หรือ), ! (ไม่)

if ให้ค่าได้

let fee = if age < 12 { 50 } else { 100 };

for: วนตามช่วงหรือตามรายการ

for i in 1..=5 {        // 1 ถึง 5
    println!("รอบที่ {i}");
}

for i in 0..3 {         // 0, 1, 2 (ไม่รวม 3)
    println!("{i}");
}

let fruits = ["มะม่วง", "ทุเรียน", "มังคุด"];
for f in fruits {
    println!("{f}");
}

for (i, f) in fruits.iter().enumerate() {   // ได้ลำดับด้วย
    println!("{}. {f}", i + 1);
}

while และ loop

let mut n = 10;
while n > 0 {
    println!("{n}");
    n -= 1;
}

let mut tries = 0;
let result = loop {        // วนไปเรื่อย ๆ จนกว่าจะ break
    tries += 1;
    if tries == 3 {
        break tries * 10;  // ออกจาก loop พร้อมคืนค่า
    }
};
println!("{result}");      // 30

break ออกจากวง, continue ข้ามไปรอบถัดไป

match: เลือกตามค่า

let day = 3;
let name = match day {
    1 => "จันทร์",
    2 => "อังคาร",
    3 => "พุธ",
    4 | 5 => "พฤหัสหรือศุกร์",
    6..=7 => "สุดสัปดาห์",
    _ => "ไม่มีวันนี้",
};

match ต้องครอบคลุมทุกค่าที่เป็นไปได้ _ หมายถึง "อย่างอื่นทั้งหมด" ถ้าลืม คอมไพเลอร์จะเตือนทันที นี่คือหนึ่งในเหตุผลที่โปรแกรม Rust บั๊กน้อย และจะเห็นพลังของ match จริง ๆ ในบทที่ 5

  1. พิมพ์สูตรคูณแม่ 2 ถึงแม่ 12 (ใช้ for ซ้อน for)
  2. FizzBuzz: พิมพ์ 1 ถึง 30 แต่ถ้าหาร 3 ลงตัวพิมพ์ Fizz หาร 5 ลงตัวพิมพ์ Buzz หารได้ทั้งสองพิมพ์ FizzBuzz (เศษจากการหารคือ %)
  3. เขียนฟังก์ชันรับคะแนน แล้วคืนเกรด A B C D F ด้วย match และช่วง 80..=100
